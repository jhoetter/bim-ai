### A — Create dgnExporter.ts

Create `packages/web/src/export/dgnExporter.ts`:

```ts
/**
 * §12.4.3: Minimal DGN export for MicroStation compatibility.
 * Produces a text-based DGN "seed" using DXF-compatible geometry with a DGN header.
 * Most DGN viewers accept this via the DXF/DGN interchange bridge.
 */

import { buildPlanView } from './dxfExporter';

export interface DgnExportOptions {
  /** Level IDs to export (all if omitted). */
  levelIds?: string[];
  /** Units: 'mm' or 'm'. Default 'mm'. */
  units?: 'mm' | 'm';
}

/**
 * Exports the scene to a DGN-compatible text format.
 * Returns a string suitable for saving as .dgn / .dxf.
 */
export function exportSceneToDgn(
  elementsById: Record<string, any>,
  levels: { id: string; name: string; elevationMm: number }[],
  options: DgnExportOptions = {},
): string {
  const { units = 'mm', levelIds } = options;
  const targetLevels = levelIds
    ? levels.filter((l) => levelIds.includes(l.id))
    : levels;

  if (targetLevels.length === 0) {
    return '; DGN SEED — no levels to export\n';
  }

  // Build DXF geometry for each level
  const planViews = targetLevels.map((level) =>
    buildPlanView(elementsById, level.id, level.name, units),
  );

  // Wrap in a DGN-compatible header comment block
  const header = [
    '; DGN SEED FILE',
    `; Generated by bim-ai — ${new Date().toISOString()}`,
    `; Levels: ${targetLevels.map((l) => l.name).join(', ')}`,
    `; Units: ${units}`,
    '; Compatible with MicroStation DGN/DXF bridge',
    '',
  ].join('\n');

  const body = planViews.join('\n\n');

  return header + body;
}

/** Returns the MIME type for DGN files. */
export const DGN_MIME_TYPE = 'application/dgn';

/** Returns a filename for the DGN export. */
export function dgnFileName(projectName: string): string {
  return `${projectName.replace(/[^a-z0-9]/gi, '_')}.dgn`;
}
```

```ts
const handleExportDgn = useCallback(() => {
  const { elementsById } = useBimStore.getState();
  const levels = Object.values(elementsById)
    .filter((e): e is any => e.kind === 'level')
    .sort((a, b) => a.elevationMm - b.elevationMm);
  const content = exportSceneToDgn(elementsById as any, levels as any);
  const blob = new Blob([content], { type: 'application/dgn' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'export.dgn';
  a.click();
  URL.revokeObjectURL(url);
}, []);
```

### E — Tests

```ts
import { describe, expect, it } from 'vitest';
import { exportSceneToDgn, dgnFileName, DGN_MIME_TYPE } from './dgnExporter';

describe('DGN exporter — §12.4.3', () => {
  it('exports a DGN seed string with header comment', () => {
    const result = exportSceneToDgn({}, [], {});
    expect(result).toContain('DGN SEED');
  });

  it('returns no-levels message when no levels', () => {
    const result = exportSceneToDgn({}, [], {});
    expect(result.length).toBeGreaterThan(0);
  });

  it('includes level names in header when levels provided', () => {
    const levels = [{ id: 'l1', name: 'EG', elevationMm: 0 }];
    const result = exportSceneToDgn({}, levels, {});
    expect(result).toContain('EG');
  });

  it('DGN_MIME_TYPE is correct', () => {
    expect(DGN_MIME_TYPE).toBe('application/dgn');
  });

  it('dgnFileName replaces spaces with underscores', () => {
    const name = dgnFileName('My Project');
    expect(name).toContain('_');
    expect(name.endsWith('.dgn')).toBe(true);
  });

  it('dgnFileName strips special characters', () => {
    const name = dgnFileName('Test/Project');
    expect(name).not.toContain('/');
    expect(name.endsWith('.dgn')).toBe(true);
  });
});
```
